function WorkspacePanel({
  user,
  onBack,
}: {
  user: User;
  onBack: () => void;
}) {
  const [workspaces, setWorkspaces] = useState<Workspace[]>([]);
  const [selectedWorkspace, setSelectedWorkspace] =
    useState<Workspace | null>(null);
  const [members, setMembers] = useState<WorkspaceMember[]>([]);
  const [settings, setSettings] = useState<WorkspaceSettings | null>(null);
  const [questions, setQuestions] = useState<WorkspaceQuestion[]>([]);
  const [selectedQuestion, setSelectedQuestion] =
    useState<WorkspaceQuestion | null>(null);
  const [attempts, setAttempts] = useState<WorkspaceAttempt[]>([]);
  const [invitations, setInvitations] =
    useState<WorkspaceInvitation[]>([]);

  const [workspaceName, setWorkspaceName] = useState("");
  const [joinCode, setJoinCode] = useState("");
  const [inviteUsername, setInviteUsername] = useState("");
  const [questionText, setQuestionText] = useState("");
  const [questionType, setQuestionType] = useState("solver");
  const [visibility, setVisibility] = useState("public");
  const [followUp, setFollowUp] = useState("");

  const [activeTab, setActiveTab] =
    useState<"dashboard" | "questions" | "members" | "invitations">(
      "dashboard",
    );

  const [busy, setBusy] = useState(false);
  const [message, setMessage] = useState("");
  const [error, setError] = useState("");

  const [createModalOpen, setCreateModalOpen] = useState(false);
  const [joinModalOpen, setJoinModalOpen] = useState(false);
  const [postModalOpen, setPostModalOpen] = useState(false);
  const [questionModalOpen, setQuestionModalOpen] = useState(false);
  const [inviteModalOpen, setInviteModalOpen] = useState(false);
  const [settingsModalOpen, setSettingsModalOpen] = useState(false);
  const [solveModalOpen, setSolveModalOpen] = useState(false);

  const [attemptModalOpen, setAttemptModalOpen] = useState(false);
  const [attemptLoading, setAttemptLoading] = useState(false);
  const [attemptId, setAttemptId] = useState<number | null>(null);
  const [attemptResponse, setAttemptResponse] =
    useState<QuestionResponse | null>(null);
  const [attemptCurrent, setAttemptCurrent] = useState(0);
  const [attemptAnswers, setAttemptAnswers] =
    useState<Record<number, number>>({});
  const [attemptHint, setAttemptHint] = useState(false);
  const [attemptFinished, setAttemptFinished] = useState(false);
  const [attemptScore, setAttemptScore] = useState<number | null>(null);

  const [solveResponse, setSolveResponse] =
    useState<QuestionResponse | null>(null);

  const isAdmin =
    selectedWorkspace?.role === "owner" ||
    selectedWorkspace?.role === "admin";

  const canSolve =
    isAdmin || settings?.allow_member_solving === true;

  async function loadWorkspaces() {
    setBusy(true);
    setError("");

    try {
      const response = await getWorkspaces();
      setWorkspaces(Array.isArray(response.data) ? response.data : []);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to load workspaces.",
      );
    } finally {
      setBusy(false);
    }
  }

  async function loadWorkspace(workspaceId: number) {
    setBusy(true);
    setError("");
    setMessage("");

    try {
      const response = await getWorkspaceDetails(workspaceId);
      const workspace = response.data as Workspace;

      setSelectedWorkspace(workspace);

      const [questionResponse, invitationResponse] =
        await Promise.all([
          getWorkspaceQuestions(workspaceId),
          getWorkspaceInvitations(),
        ]);

      setQuestions(
        Array.isArray(questionResponse.data)
          ? questionResponse.data
          : [],
      );

      setInvitations(
        Array.isArray(invitationResponse.data)
          ? invitationResponse.data
          : [],
      );

      const detailedWorkspace = workspace as Workspace & {
        members?: WorkspaceMember[];
        settings?: WorkspaceSettings;
      };

      setMembers(detailedWorkspace.members ?? []);
      setSettings(detailedWorkspace.settings ?? null);

      setSelectedQuestion(null);
      setSolveResponse(null);
      setAttempts([]);
      setQuestionModalOpen(false);
      setSolveModalOpen(false);
      setAttemptModalOpen(false);
      setActiveTab("dashboard");
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to load workspace.",
      );
    } finally {
      setBusy(false);
    }
  }

  async function createNewWorkspace() {
    const name = workspaceName.trim();

    if (!name) return;

    setBusy(true);
    setError("");
    setMessage("");

    try {
      const response = await createWorkspace(name);
      const workspace = response.data as Workspace;

      setWorkspaceName("");
      setCreateModalOpen(false);
      setMessage("Workspace created successfully.");

      await loadWorkspaces();

      if (workspace?.id != null) {
        await loadWorkspace(Number(workspace.id));
      }
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to create workspace.",
      );
    } finally {
      setBusy(false);
    }
  }

  async function joinExistingWorkspace() {
    const code = joinCode.trim();

    if (!code) return;

    setBusy(true);
    setError("");
    setMessage("");

    try {
      const response = await joinWorkspace(code);
      const workspace = response.data as Workspace;

      setJoinCode("");
      setJoinModalOpen(false);
      setMessage("Joined workspace successfully.");

      await loadWorkspaces();

      if (workspace?.id != null) {
        await loadWorkspace(Number(workspace.id));
      }
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to join workspace.",
      );
    } finally {
      setBusy(false);
    }
  }

  async function refreshQuestions() {
    if (!selectedWorkspace?.id) return;

    try {
      const response = await getWorkspaceQuestions(
        Number(selectedWorkspace.id),
      );

      setQuestions(
        Array.isArray(response.data)
          ? response.data
          : [],
      );
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to load questions.",
      );
    }
  }

  async function createQuestion() {
    if (!selectedWorkspace?.id || !questionText.trim()) {
      return;
    }

    setBusy(true);
    setError("");
    setMessage("");

    try {
      await createWorkspaceQuestion({
        WorkspaceId: Number(selectedWorkspace.id),
        Question: questionText.trim(),
        QuestionType: questionType,
        Visibility: visibility,
      });

      setQuestionText("");
      setPostModalOpen(false);
      setMessage("Question posted successfully.");

      await refreshQuestions();
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to create question.",
      );
    } finally {
      setBusy(false);
    }
  }

  async function openQuestion(questionId: number) {
    if (!selectedWorkspace?.id) return;

    setBusy(true);
    setError("");
    setMessage("");

    try {
      const response = await getWorkspaceQuestionDetails(
        Number(selectedWorkspace.id),
        questionId,
      );

      setSelectedQuestion(response.data as WorkspaceQuestion);

      try {
        const attemptResponse =
          await getWorkspaceQuestionAttempts(
            Number(selectedWorkspace.id),
            questionId,
          );

        setAttempts(
          Array.isArray(attemptResponse.data)
            ? attemptResponse.data
            : [],
        );
      } catch {
        setAttempts([]);
      }

      setAttemptFinished(false);
      setAttemptScore(null);
      setSolveResponse(null);
      setQuestionModalOpen(true);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to open question.",
      );
    } finally {
      setBusy(false);
    }
  }

  async function computeQuestion() {
    if (!selectedQuestion?.question || !canSolve) return;

    setAttemptLoading(true);
    setError("");
    setMessage("");

    try {
      const rawResponse = await postQuestion(
        user.username,
        selectedQuestion.question,
      );

      const response = validateQuestionResponse(rawResponse);

      if (!response || response.is_relevant !== true) {
        throw new Error(
          response?.error_message ||
            "Gemini could not process this Workspace question.",
        );
      }

      setSolveResponse(response);
      setQuestionModalOpen(false);
      setSolveModalOpen(true);
      setMessage("Gemini solution prepared.");
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Gemini computation failed.",
      );
    } finally {
      setAttemptLoading(false);
    }
  }

  async function createAttempt() {
    if (
      !selectedWorkspace?.id ||
      !selectedQuestion?.id ||
      !canSolve
    ) {
      return;
    }

    setAttemptLoading(true);
    setError("");
    setMessage("");
    setAttemptFinished(false);
    setAttemptScore(null);
    setAttemptAnswers({});
    setAttemptCurrent(0);
    setAttemptHint(false);
    setAttemptResponse(null);

    try {
      const attempt = await createWorkspaceAttempt(
        Number(selectedWorkspace.id),
        Number(selectedQuestion.id),
        "solver",
        "",
      );

      const createdAttempt =
        attempt.data as WorkspaceAttempt;

      if (createdAttempt?.id != null) {
        setAttemptId(Number(createdAttempt.id));
      }

      const rawResponse = await postQuestion(
        user.username,
        selectedQuestion.question,
      );

      const response = validateQuestionResponse(rawResponse);

      if (!response || response.is_relevant !== true) {
        throw new Error(
          response?.error_message ||
            "Gemini could not prepare this attempt.",
        );
      }

      if (
        !response.ai_questions ||
        response.ai_questions.length === 0
      ) {
        throw new Error(
          "Gemini did not return a usable practice session.",
        );
      }

      setAttemptResponse(response);
      setQuestionModalOpen(false);
      setAttemptModalOpen(true);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to start the Workspace attempt.",
      );
    } finally {
      setAttemptLoading(false);
    }
  }

  function chooseWorkspaceAttemptAnswer(option: number) {
    if (attemptFinished) return;

    setAttemptAnswers((previous) => ({
      ...previous,
      [attemptCurrent]: option,
    }));
  }

  async function finishWorkspaceAttempt() {
    if (
      !selectedWorkspace?.id ||
      attemptId == null ||
      !attemptResponse
    ) {
      return;
    }

    const generatedQuestions =
      attemptResponse.ai_questions ?? [];

    const finalScore = generatedQuestions.reduce(
      (total, item, index) =>
        total +
        (attemptAnswers[index] === item.correct_option
          ? 1
          : 0),
      0,
    );

    setAttemptScore(finalScore);
    setAttemptFinished(true);
    setAttemptHint(false);

    try {
      await completeWorkspaceAttempt(
        Number(selectedWorkspace.id),
        Number(attemptId),
        JSON.stringify(attemptAnswers),
        finalScore,
        true,
      );

      if (selectedQuestion?.id != null) {
        const response =
          await getWorkspaceQuestionAttempts(
            Number(selectedWorkspace.id),
            Number(selectedQuestion.id),
          );

        setAttempts(
          Array.isArray(response.data)
            ? response.data
            : [],
        );
      }

      setMessage("Attempt completed and recorded.");
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "The result was shown, but the attempt could not be recorded.",
      );
    }
  }

  function nextWorkspaceAttemptQuestion() {
    if (!attemptResponse || attemptFinished) return;

    const total = attemptResponse.ai_questions.length;

    if (attemptAnswers[attemptCurrent] == null) {
      return;
    }

    if (attemptCurrent >= total - 1) {
      void finishWorkspaceAttempt();
      return;
    }

    setAttemptHint(false);
    setAttemptCurrent((current) => current + 1);
  }

  async function sendFollowUp() {
    if (
      !selectedWorkspace?.id ||
      !selectedQuestion?.id ||
      !attemptFinished ||
      !followUp.trim()
    ) {
      return;
    }

    setBusy(true);
    setError("");
    setMessage("");

    try {
      await postWorkspaceQuestionFollowUp(
        Number(selectedWorkspace.id),
        Number(selectedQuestion.id),
        followUp.trim(),
      );

      setFollowUp("");
      setMessage("Follow-up question submitted.");
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to submit follow-up.",
      );
    } finally {
      setBusy(false);
    }
  }

  async function sendInvitation() {
    if (
      !selectedWorkspace?.id ||
      !inviteUsername.trim()
    ) {
      return;
    }

    setBusy(true);
    setError("");
    setMessage("");

    try {
      await inviteToWorkspace(
        Number(selectedWorkspace.id),
        inviteUsername.trim(),
      );

      setInviteUsername("");
      setInviteModalOpen(false);
      setMessage("Invitation sent.");
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to send invitation.",
      );
    } finally {
      setBusy(false);
    }
  }

  async function handleAcceptInvitation(id: number) {
    setBusy(true);
    setError("");

    try {
      await acceptWorkspaceInvitation(id);

      const response =
        await getWorkspaceInvitations();

      setInvitations(
        Array.isArray(response.data)
          ? response.data
          : [],
      );

      await loadWorkspaces();
      setMessage("Invitation accepted.");
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to accept invitation.",
      );
    } finally {
      setBusy(false);
    }
  }

  async function handleDeclineInvitation(id: number) {
    setBusy(true);
    setError("");

    try {
      await declineWorkspaceInvitation(id);

      const response =
        await getWorkspaceInvitations();

      setInvitations(
        Array.isArray(response.data)
          ? response.data
          : [],
      );

      setMessage("Invitation declined.");
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to decline invitation.",
      );
    } finally {
      setBusy(false);
    }
  }

  async function leaveCurrentWorkspace() {
    if (!selectedWorkspace?.id) return;

    setBusy(true);
    setError("");

    try {
      await leaveWorkspace(
        Number(selectedWorkspace.id),
      );

      setSelectedWorkspace(null);
      setQuestions([]);
      setMembers([]);
      setSettings(null);
      setSelectedQuestion(null);
      setAttempts([]);
      setQuestionModalOpen(false);
      setSolveModalOpen(false);
      setAttemptModalOpen(false);

      await loadWorkspaces();
      setMessage("You left the workspace.");
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Unable to leave workspace.",
      );
    } finally {
      setBusy(false);
    }
  }

  useEffect(() => {
    void loadWorkspaces();
  }, []);

  useEffect(() => {
    if (!selectedWorkspace?.id) return;

    const workspace = selectedWorkspace as Workspace & {
      members?: WorkspaceMember[];
      settings?: WorkspaceSettings;
    };

    if (workspace.members) {
      setMembers(workspace.members);
    }

    if (workspace.settings) {
      setSettings(workspace.settings);
    }
  }, [selectedWorkspace]);


  return (
    <motion.div
      className="workspace-page workspace-v4"
      initial={{ opacity: 0, y: 18 }}
      animate={{ opacity: 1, y: 0 }}
      transition={{ duration: 0.35 }}
    >
      <div className="workspace-header">
        <div>
          <span className="eyebrow">COLLABORATIVE LEARNING</span>
          <h1>Workspaces</h1>
          <p>
            Work through computational problems together
            with Gemini-powered solving and guided attempts.
          </p>
        </div>

        <div className="workspace-header-actions">
          <span className="workspace-online-badge">
            <i />
            Online Â· Gemini
          </span>

          <button
            type="button"
            className="secondary"
            onClick={onBack}
          >
            Back to Solver
          </button>
        </div>
      </div>

      {error && (
        <div className="error review-error workspace-alert">
          {error}
        </div>
      )}

      {message && (
        <div className="workspace-message workspace-alert">
          {message}
        </div>
      )}

      {!selectedWorkspace ? (
        <div className="workspace-landing-grid">
          <section className="workspace-hero-card">
            <div className="workspace-hero-mark">
              W
            </div>

            <span className="eyebrow">
              YOUR COLLABORATION SPACE
            </span>

            <h2>
              Learn, solve and collaborate
              <br />
              in one place.
            </h2>

            <p>
              Create a private workspace or join one using
              an invitation code. Questions, attempts and
              progress stay organised inside the workspace.
            </p>

            <div className="workspace-hero-actions">
              <button
                type="button"
                className="primary"
                onClick={() => setCreateModalOpen(true)}
              >
                Create Workspace
              </button>

              <button
                type="button"
                className="secondary"
                onClick={() => setJoinModalOpen(true)}
              >
                Join Workspace
              </button>
            </div>
          </section>

          <section className="workspace-list-card workspace-v4-card">
            <div className="workspace-card-header">
              <div>
                <span className="eyebrow">SPACES</span>
                <h2>Your Workspaces</h2>
                <span>
                  {workspaces.length} active
                  {workspaces.length === 1
                    ? " workspace"
                    : " workspaces"}
                </span>
              </div>

              <button
                type="button"
                className="secondary workspace-icon-button"
                onClick={() => void loadWorkspaces()}
                disabled={busy}
                aria-label="Refresh workspaces"
              >
                â†»
              </button>
            </div>

            <div className="workspace-list workspace-v4-list">
              {workspaces.length === 0 ? (
                <div className="workspace-empty workspace-empty-large">
                  <div className="workspace-empty-icon">
                    W
                  </div>
                  <strong>No workspaces yet</strong>
                  <span>
                    Create one or join an existing workspace
                    to get started.
                  </span>
                </div>
              ) : (
                workspaces.map((workspace) => (
                  <button
                    type="button"
                    key={workspace.id}
                    className="workspace-list-item workspace-v4-list-item"
                    onClick={() =>
                      void loadWorkspace(
                        Number(workspace.id),
                      )
                    }
                  >
                    <span className="workspace-list-icon">
                      {String(workspace.name || "W")
                        .trim()
                        .charAt(0)
                        .toUpperCase() || "W"}
                    </span>

                    <span className="workspace-list-copy">
                      <strong>{workspace.name}</strong>
                      <small>
                        {workspace.role ?? "member"}
                      </small>
                    </span>

                    <span className="workspace-list-arrow">
                      â†’
                    </span>
                  </button>
                ))
              )}
            </div>
          </section>
        </div>
      ) : (
        <div className="workspace-shell workspace-v4-shell">
          <aside className="workspace-sidebar workspace-v4-sidebar">
            <div className="workspace-sidebar-heading">
              <span className="workspace-list-icon">
                {String(selectedWorkspace.name || "W")
                  .trim()
                  .charAt(0)
                  .toUpperCase() || "W"}
              </span>

              <div>
                <strong>{selectedWorkspace.name}</strong>
                <small>
                  {selectedWorkspace.role ?? "member"}
                </small>
              </div>
            </div>

            <div className="workspace-sidebar-status">
              <span className="workspace-status-dot" />
              Online Â· Gemini
            </div>

            <nav className="workspace-sidebar-nav">
              {(
                [
                  ["dashboard", "Overview"],
                  ["questions", "Questions"],
                  ["members", "Members"],
                  ["invitations", "Invitations"],
                ] as const
              ).map(([tab, label]) => (
                <button
                  type="button"
                  key={tab}
                  className={
                    activeTab === tab ? "active" : ""
                  }
                  onClick={() => setActiveTab(tab)}
                >
                  <span>
                    {tab === "dashboard"
                      ? "âŒ‚"
                      : tab === "questions"
                        ? "?"
                        : tab === "members"
                          ? "â—Ž"
                          : "âœ‰"}
                  </span>
                  {label}
                  {tab === "questions" && (
                    <small>{questions.length}</small>
                  )}
                  {tab === "members" && (
                    <small>{members.length}</small>
                  )}
                  {tab === "invitations" &&
                    invitations.length > 0 && (
                      <small>{invitations.length}</small>
                    )}
                </button>
              ))}
            </nav>

            <div className="workspace-sidebar-bottom">
              <button
                type="button"
                className="secondary"
                onClick={() => setSettingsModalOpen(true)}
              >
                Settings
              </button>

              <button
                type="button"
                className="secondary danger-action"
                onClick={() =>
                  void leaveCurrentWorkspace()
                }
                disabled={busy}
              >
                Leave Workspace
              </button>
            </div>
          </aside>

          <main className="workspace-main workspace-v4-main">
            {activeTab === "dashboard" && (
              <div className="workspace-dashboard workspace-v4-dashboard">
                <div className="workspace-dashboard-heading">
                  <div>
                    <span className="eyebrow">
                      WORKSPACE OVERVIEW
                    </span>
                    <h2>Ready to work?</h2>
                    <p>
                      Pick a question, solve it with Gemini,
                      or start a guided attempt.
                    </p>
                  </div>

                  <button
                    type="button"
                    className="primary"
                    onClick={() => setActiveTab("questions")}
                  >
                    Browse Questions
                  </button>
                </div>

                <div className="workspace-stat-grid workspace-v4-stats">
                  <div className="workspace-stat">
                    <span>Questions</span>
                    <strong>{questions.length}</strong>
                    <small>Posted in this workspace</small>
                  </div>

                  <div className="workspace-stat">
                    <span>Members</span>
                    <strong>{members.length}</strong>
                    <small>People collaborating</small>
                  </div>

                  <div className="workspace-stat">
                    <span>Invitations</span>
                    <strong>{invitations.length}</strong>
                    <small>Waiting for response</small>
                  </div>

                  <div className="workspace-stat">
                    <span>Solver</span>
                    <strong>Gemini</strong>
                    <small>Online computation</small>
                  </div>
                </div>

                <div className="workspace-command-grid">
                  <button
                    type="button"
                    className="workspace-command-card"
                    onClick={() => {
                      setActiveTab("questions");
                      setPostModalOpen(true);
                    }}
                  >
                    <span>+</span>
                    <strong>Post a Question</strong>
                    <small>
                      Add a problem for the workspace.
                    </small>
                  </button>

                  <button
                    type="button"
                    className="workspace-command-card"
                    onClick={() => setActiveTab("questions")}
                  >
                    <span>?</span>
                    <strong>Open Questions</strong>
                    <small>
                      Browse and attempt workspace problems.
                    </small>
                  </button>

                  <button
                    type="button"
                    className="workspace-command-card"
                    onClick={() => setInviteModalOpen(true)}
                  >
                    <span>+</span>
                    <strong>Invite Member</strong>
                    <small>
                      Bring another user into the workspace.
                    </small>
                  </button>

                  {isAdmin && (
                    <button
                      type="button"
                      className="workspace-command-card"
                      onClick={() =>
                        setSettingsModalOpen(true)
                      }
                    >
                      <span>âš™</span>
                      <strong>Workspace Controls</strong>
                      <small>
                        Decide who can post and solve.
                      </small>
                    </button>
                  )}
                </div>

                <div className="workspace-recent-card">
                  <div className="workspace-card-header">
                    <div>
                      <span className="eyebrow">
                        RECENT QUESTIONS
                      </span>
                      <h3>Continue learning</h3>
                    </div>

                    <button
                      type="button"
                      className="secondary"
                      onClick={() =>
                        setActiveTab("questions")
                      }
                    >
                      View all
                    </button>
                  </div>

                  {questions.length === 0 ? (
                    <div className="workspace-empty">
                      No questions have been posted yet.
                    </div>
                  ) : (
                    <div className="workspace-recent-list">
                      {questions.slice(0, 4).map((question) => (
                        <button
                          type="button"
                          className="workspace-recent-item"
                          key={question.id}
                          onClick={() =>
                            void openQuestion(
                              Number(question.id),
                            )
                          }
                        >
                          <span className="workspace-question-number">
                            Q
                          </span>

                          <span>
                            <strong>
                              {question.question}
                            </strong>
                            <small>
                              {question.question_type ??

                      className="secondary"

                        setAttemptModalOpen(false);
                        setQuestionModalOpen(true);
                      }}
                    >
                      Review Question
                    </button>

                    <button
                      type="button"
                      className="primary"
                      onClick={() =>
                        setAttemptModalOpen(false)
                      }
                    >
                      Return to Workspace
                    </button>
                  </div>
                </div>
              ) : (
                <>
                  {attemptResponse.ai_questions[
                    attemptCurrent
                  ] && (
                    <>
                      <div className="workspace-attempt-progress">
                        <div>
                          <span>
                            Question{" "}
                            {attemptCurrent + 1}
                          </span>
                          <small>
                            of{" "}
                            {
                              attemptResponse
                                .ai_questions.length
                            }
                          </small>
                        </div>

                        <div className="workspace-attempt-progress-track">
                          <i
                            style={{
                              width: `${
                                ((attemptCurrent + 1) /
                                  attemptResponse
                                    .ai_questions.length) *
                                100
                              }%`,
                            }}
                          />
                        </div>
                      </div>

                      <div className="workspace-attempt-question workspace-v4-attempt-question">
                        <span className="workspace-question-label">
                          QUESTION{" "}
                          {String(
                            attemptCurrent + 1,
                          ).padStart(2, "0")}
                        </span>

                        <MathText>
                          {
                            attemptResponse
                              .ai_questions[
                              attemptCurrent
                            ].question
                          }
                        </MathText>
                      </div>

                      <div className="workspace-attempt-options workspace-v4-attempt-options">
                        {attemptResponse.ai_questions[
                          attemptCurrent
                        ].options.map(
                          (option, index) => {
                            const selected =
                              attemptAnswers[
                                attemptCurrent
                              ] === index;

                            return (
                              <button
                                type="button"
                                key={index}
                                className={
                                  selected
                                    ? "selected"
                                    : ""
                                }
                                onClick={() =>
                                  chooseWorkspaceAttemptAnswer(
                                    index,
                                  )
                                }
                              >
                                <span className="workspace-option-letter">
                                  {String.fromCharCode(
                                    65 + index,
                                  )}
                                </span>

                                <span className="workspace-option-content">
                                  <MathText>
                                    {option}
                                  </MathText>
                                </span>

                                <span className="workspace-option-check">
                                  {selected ? "âœ“" : ""}
                                </span>
                              </button>
                            );
                          },
                        )}
                      </div>

                      {attemptHint && (
                        <div className="workspace-attempt-hint workspace-v4-hint">
                          <strong>Hint</strong>
                          <p>
                            {
                              attemptResponse
                                .ai_questions[
                                attemptCurrent
                              ].hint
                            }
                          </p>
                        </div>
                      )}

                      <div className="workspace-attempt-window-actions">
                        <button
                          type="button"
                          className="secondary"
                          onClick={() =>
                            setAttemptHint(
                              (value) => !value,
                            )
                          }
                        >
                          {attemptHint
                            ? "Hide Hint"
                            : "Show Hint"}
                        </button>

                        <button
                          type="button"
                          className="primary"
                          disabled={
                            attemptAnswers[
                              attemptCurrent
                            ] == null
                          }
                          onClick={
                            nextWorkspaceAttemptQuestion
                          }
                        >
                          {attemptCurrent ===
                          attemptResponse.ai_questions
                            .length -
                            1
                            ? "Finish Attempt"
                            : "Next Question â†’"}
                        </button>
                      </div>
                    </>
                  )}
                </>
              )}
            </motion.div>
          </motion.div>
        )}
      </AnimatePresence>
    </motion.div>
  );
}
function WorkspaceSettingsPanel({
  workspace,
  settings,
  enabled,
  busy,
  onSaved,
  onError,
}: {
  workspace: Workspace;
  settings: WorkspaceSettings | null;
  enabled: boolean;
  busy: boolean;
  onSaved: (settings: WorkspaceSettings) => void;
  onError: (message: string) => void;
}) {
  const [mode, setMode] = useState<"offline" | "online">(settings?.computation_mode === "online" ? "online" : "offline");
  const [allowPosting, setAllowPosting] = useState(
    settings?.allow_member_posting ?? true,
  );
  const [allowSolving, setAllowSolving] = useState(
    settings?.allow_member_solving ?? true,
  );

  useEffect(() => {
    setMode(settings?.computation_mode === "online" ? "online" : "offline");
    setAllowPosting(settings?.allow_member_posting ?? true);
    setAllowSolving(settings?.allow_member_solving ?? true);
  }, [settings]);

  async function save() {
    if (!enabled || workspace.id == null) return;

    try {
      const response = await updateWorkspaceSettings({
        WorkspaceId: Number(workspace.id),
        ComputationMode: mode,
        AllowMemberPosting: allowPosting,
        AllowMemberSolving: allowSolving,
      });

      onSaved(response.data as WorkspaceSettings);
    } catch (err) {
      onError(
        err instanceof Error
          ? err.message
          : "Unable to update workspace settings.",
      );
    }
  }

  return (
    <div className="workspace-settings">
      <div className="workspace-card-header">
        <div>
          <h3>Workspace Settings</h3>
          <span>
            {enabled
              ? "Owner/admin controls"
              : "Only workspace owners and admins can change these settings."}
          </span>
        </div>
      </div>

      <label className="workspace-setting">
        <span>Computation mode</span>
        <select
          value={mode}
          onChange={(event) =>
            setMode(event.target.value as "offline" | "online")
          }
          disabled={!enabled}
        >
          <option value="offline">Offline</option>
          <option value="online">Online</option>
        </select>
      </label>

      <label className="workspace-setting-toggle">
        <input
          type="checkbox"
          checked={allowPosting}
          onChange={(event) => setAllowPosting(event.target.checked)}
          disabled={!enabled}
        />
        <span>Allow members to post questions</span>
      </label>

      <label className="workspace-setting-toggle">
        <input
          type="checkbox"
          checked={allowSolving}
          onChange={(event) => setAllowSolving(event.target.checked)}
          disabled={!enabled}
        />
        <span>Allow members to solve questions</span>
      </label>

      <button
        type="button"
        className="primary"
        onClick={() => void save()}
        disabled={!enabled || busy}
      >
        Save Settings
      </button>
    </div>
  );
}
